    <script>
        function launchConfetti() {
            // Safety check: do nothing (and don't throw) if the library didn't load
            if (typeof confetti !== 'function') {
                console.warn('canvas-confetti library did not load');
                return;
            }

            var duration = 3 * 1000;
            var animationEnd = Date.now() + duration;
            // zIndex is above the popup overlay (999999) so confetti isn't hidden behind it
            var defaults = { startVelocity: 30, spread: 360, ticks: 60, zIndex: 1000001 };

            function randomInRange(min, max) {
                return Math.random() * (max - min) + min;
            }

            (function frame() {
                confetti({
                    particleCount: 10,
                    angle: randomInRange(55, 125),
                    spread: 70,
                    origin: { x: Math.random(), y: Math.random() - 0.2 },
                    ...defaults
                });

                if (Date.now() < animationEnd) {
                    requestAnimationFrame(frame);
                }
            })();
        }
    </script>
